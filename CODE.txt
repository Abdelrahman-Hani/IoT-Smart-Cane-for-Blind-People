int trigPin = 11;   // الأردوينو بيبعت منه إشارة للحساس (يبعت نبضة)
int echoPin = 8;    // الأردوينو بيستقبل عليه الإشارة الراجعة من الحساس

int buzzer = 2;     // البازر (بيطلع صوت)
int led = 5;       // الليد (بينور)
// =======================
// Variables
// =======================
long duration;   // الوقت اللي الموجة خده رايح وجاي
int distance;    // المسافة اللي هنحسبها

// =======================
// Setup (مرة واحدة بس)
// =======================
void setup() {
  Serial.begin(9600);  
  // ده بيخلي الأردوينو يتكلم مع الكمبيوتر ويطبع عليه

  pinMode(trigPin, OUTPUT); 
  // هنا الأردوينو هو اللي بيبعت إشارة للحساس
  pinMode(echoPin, INPUT);  
  // هنا الأردوينو بيستقبل الإشارة اللي راجعة من الحساس

  pinMode(buzzer, OUTPUT);  
  // الأردوينو بيشغل البازر

  pinMode(led, OUTPUT);     
  // الأردوينو بينور الليد
}

// =======================
// Loop (بيتكرر طول الوقت)
// =======================
void loop() {

  // 🔥 إرسال نبضة للحساس
  digitalWrite(trigPin, LOW);  
  // الأول بنخليه 0 عشان نبدأ نظيف

  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH); 
  // هنا الأردوينو بيبعت نبضة للحساس (كأنها "ابدأ قيس")

  delayMicroseconds(10);       
  // النبضة بتفضل شغالة 10 ميكروثانية

  digitalWrite(trigPin, LOW);  
  // وقف الإشارة بعد ما بعتناها

  // 📡 استقبال الإشارة من الحساس
  duration = pulseIn(echoPin, HIGH);
  // هنا الأردوينو بيستنى الحساس يرد ويرجع الإشارة
  // وبيقيس الوقت اللي الموجة خده رايح وجاي

  // 📏 حساب المسافة
  distance = duration * 0.034 / 2;
  // نحول الوقت لمسافة (الصوت رايح وجاي)
  // 🖥️ طباعة النتيجة على الكمبيوتر
  Serial.print("Distance: ");
  Serial.println(distance);

  // 🚨 قرار: قريب ولا بعيد
  if (distance <= 100) {
    // لو الحاجة قريبة

    digitalWrite(led, HIGH);  
    // نور الليد

    digitalWrite(buzzer, HIGH); 
    // شغل البازر (صوت)
  } 
  else {
    // لو الحاجة بعيدة

    digitalWrite(led, LOW);   
    // طفي الليد

    digitalWrite(buzzer, LOW); 
    // طفي البازر
  }

  delay(200); // استنى شوية وكرر
}
